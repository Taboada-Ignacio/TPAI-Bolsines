import React, { useState, useEffect } from "react";

export interface SplitFlapProps {
  text: string;
  size?: "sm" | "lg";
  className?: string;
}

/**
 * SplitFlap
 * Emula los paneles mecánicos de aeropuertos de Solari di Udine.
 * Rota caracteres a 45ms en verde lima flúor (#d4ff3a) antes de clavar en la letra final.
 */
export function SplitFlap({ text, size = "lg", className = "" }: SplitFlapProps) {
  const chars = text
    .toUpperCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .split("");

  return (
    <span className={`inline-flex flex-wrap gap-[3px] select-none ${className}`} aria-label={text}>
      {chars.map((char, index) => (
        <SplitFlapChar key={`${index}-${char}`} char={char} delay={index * 40} size={size} />
      ))}
    </span>
  );
}

function SplitFlapChar({ char, delay = 0, size = "lg" }: { char: string; delay?: number; size?: "sm" | "lg" }) {
  const [current, setCurrent] = useState(char);
  const isFlipping = current !== char;

  useEffect(() => {
    let count = 0;
    const max = 6 + Math.floor(delay / 40);
    const alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789·";
    const timer = setInterval(() => {
      count++;
      if (count >= max) {
        setCurrent(char);
        clearInterval(timer);
      } else {
        setCurrent(alphabet[Math.floor(Math.random() * alphabet.length)]);
      }
    }, 45);

    return () => clearInterval(timer);
  }, [char, delay]);

  const sizeClass =
    size === "lg"
      ? "h-[1.5em] w-[1.05em] text-[clamp(1.4rem,2.6vw,2.2rem)]"
      : "h-[1.6em] w-[1.15em] text-[14px]";

  return (
    <span
      className={`relative inline-grid place-items-center overflow-hidden rounded-[2px] bg-[#1c1c1c] font-mono font-semibold ${sizeClass}`}
    >
      <span
        className={`transition-all duration-75 ${
          isFlipping ? "-translate-y-[3%] text-[#d4ff3a]" : "text-[#f5f5f5]"
        }`}
      >
        {current === " " ? "\u00A0" : current}
      </span>
      {/* Hendidura mecánica de 1px en el centro */}
      <span className="absolute inset-x-0 top-1/2 h-px bg-black/60 pointer-events-none" />
    </span>
  );
}
